class Solution {
    public int[] deckRevealedIncreasing(int[] deck) {
        Arrays.sort(deck);
        Queue<Integer> q=new LinkedList<>();
        for(int i=0;i<deck.length;i++){
            q.add(i);
        }
        int[] a=new int[deck.length];
        for(int b:deck){
            a[q.poll()]=b;
            if(!q.isEmpty()){
                q.add(q.poll());
            }
        }
        return a;
    }
}
